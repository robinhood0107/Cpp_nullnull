cmake_minimum_required(VERSION 3.28)
project(Cpp_nullnull LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
include(CTest)

if(MSVC)
    add_compile_options(/utf-8 /W4 /permissive- /Zc:__cplusplus)
    add_compile_definitions(_CRT_SECURE_NO_WARNINGS NOMINMAX)
else()
    add_compile_options(-Wall -Wextra -Wpedantic)
endif()

include(cmake/Terminal.cmake)
file(READ "${CMAKE_SOURCE_DIR}/docs/targets.json" registry)
string(JSON count LENGTH "${registry}")
math(EXPR last "${count} - 1")
foreach(index RANGE ${last})
    string(JSON name GET "${registry}" ${index} name)
    string(JSON directory GET "${registry}" ${index} path)
    string(JSON source_count LENGTH "${registry}" ${index} sources)
    math(EXPR source_last "${source_count} - 1")
    set(sources)
    foreach(source_index RANGE ${source_last})
        string(JSON source GET "${registry}" ${index} sources ${source_index})
        list(APPEND sources "${directory}/${source}")
    endforeach()
    add_executable(${name} ${sources})
    if(name MATCHES "^assignment0[2345]$")
        target_link_libraries(${name} PRIVATE terminal)
    endif()
endforeach()

add_subdirectory(projects/my-string)
if(BUILD_TESTING)
    add_subdirectory(tests)
endif()
